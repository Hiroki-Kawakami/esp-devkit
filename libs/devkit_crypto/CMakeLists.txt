# Cryptography: hashes, MACs, KDFs, ciphers, AEADs, key agreement, signatures,
# X.509 and TLS 1.3.
set(requires devkit idf rust_rt)
if(ESP_PLATFORM)
    list(APPEND requires mbedtls)
endif()
idf_component_register(SRCS "/dev/null" PRIV_REQUIRES ${requires})
if(NOT ESP_PLATFORM)
    target_include_directories(${COMPONENT_LIB} PRIVATE ${TFPSACRYPTO_INCLUDE_DIRS})
endif()

devkit_rust_out_dir(out_dir)
set(bundle_rs "")
set(bundle_depends "")
if(CONFIG_DEVKIT_CRYPTO_CA_BUNDLE_EMBEDDED OR CONFIG_DEVKIT_CRYPTO_CA_BUNDLE_PARTITION)
    set(bundle_dir "$ENV{DEVKIT_CA_BUNDLE}")
    if(NOT EXISTS "${bundle_dir}/ca_bundle.bin")
        message(FATAL_ERROR "devkit_crypto: DEVKIT_CA_BUNDLE is not set; configure from the dev shell (nix develop)")
    endif()
    set(bundle_depends "${bundle_dir}/ca_bundle.bin" "${bundle_dir}/ca_partition.bin")
endif()
if(CONFIG_DEVKIT_CRYPTO_CA_BUNDLE_EMBEDDED)
    set(bundle_rs "pub(crate) static BUNDLE: &[u8] = include_bytes!(\"${bundle_dir}/ca_bundle.bin\");\n")
elseif(CONFIG_DEVKIT_CRYPTO_CA_BUNDLE_PARTITION)
    set(bundle_rs "pub(crate) const IMAGE: &str = \"${bundle_dir}/ca_partition.bin\";\n")
    if(ESP_PLATFORM)
        partition_table_get_partition_info(size "--partition-name ${CONFIG_DEVKIT_CRYPTO_CA_PARTITION}" "size")
        if(NOT size)
            message(FATAL_ERROR "devkit_crypto: no partition '${CONFIG_DEVKIT_CRYPTO_CA_PARTITION}' in the partition table")
        endif()
        # esptool keeps a copy of what it flashed next to the image, which the Nix store does not allow.
        file(COPY_FILE "${bundle_dir}/ca_partition.bin" "${out_dir}/ca_partition.bin" ONLY_IF_DIFFERENT)
        esp_partition_register_target(${CONFIG_DEVKIT_CRYPTO_CA_PARTITION} "${out_dir}/ca_partition.bin" FLASH_IN_PROJECT)
    endif()
endif()
file(CONFIGURE OUTPUT "${out_dir}/ca_bundle.rs" CONTENT "${bundle_rs}" @ONLY)

devkit_rust_crate(ROOT "src/lib.rs" TESTS OPT_LEVEL 2 BINDGEN_HEADERS "psa/crypto.h"
    DEPENDS "${out_dir}/ca_bundle.rs" ${bundle_depends})
